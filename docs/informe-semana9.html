4. Cambios aplicados (DIP + DI)

4.1 Contrato estructural — src/aplicacion/contratos/repositorios.js (nuevo)

function exigirMetodos(objeto, metodos, nombre) {
  for (const metodo of metodos) {
    if (typeof objeto?.[metodo] !== 'function') {
      throw new TypeError(`${nombre} debe implementar ${metodo}()`);
    }
  }
}
function verificarRepositorios({ citaRepository, profesionalRepository }) {
  exigirMetodos(citaRepository,
    ['listarTodas', 'existeEnHorario', 'guardar'], 'citaRepository');
  exigirMetodos(profesionalRepository, ['listarTodos'], 'profesionalRepository');
}
module.exports = { verificarRepositorios };

4.2 Servicio de aplicación — src/aplicacion/citaService.js

Se eliminaron los require('../persistencia/...'). El servicio ahora es una factoría que recibe los repositorios y un reloj inyectado para pruebas reproducibles.

const reglas = require('../dominio/reglasDeAgenda');
const { verificarRepositorios } = require('./contratos/repositorios');

function crearCitaService({ citaRepository, profesionalRepository,
                            reloj = () => new Date() }) {
  verificarRepositorios({ citaRepository, profesionalRepository });
  return {
    consultarCitas: () => citaRepository.listarTodas(),
    consultarProfesionales: () => profesionalRepository.listarTodos(),
    reservarCita: async (datos) => {
      reglas.validarDatosCompletos(datos);
      reglas.validarFechaFutura(datos.fecha_hora, reloj());
      const ocupado = await citaRepository.existeEnHorario(
        datos.profesional_id, datos.fecha_hora);
      reglas.validarAgendaLibre(ocupado);
      const id = await citaRepository.guardar(datos);
      return { mensaje: 'Cita creada', id };
    }
  };
}
module.exports = { crearCitaService };

4.4 Punto de composición — server.js

Único lugar donde se importan implementaciones concretas y se inyectan.

const { crearCitaService } = require('./src/aplicacion/citaService');
const citaRepository = require('./src/persistencia/citaRepository');
const profesionalRepository = require('./src/persistencia/profesionalRepository');

const citaService = crearCitaService({ citaRepository, profesionalRepository });

10. Conclusiones

  • La capa de aplicación pasó de depender de módulos concretos a depender de un contrato (DIP).
  • Las dependencias se construyen e inyectan en server.js, el punto de composición (DI).
  • El reloj inyectado permite pruebas deterministas de la regla de fecha futura.
  • Las 4 pruebas unitarias se ejecutan sin base de datos ni red.
  • Los endpoints de la API se conservan sin cambios funcionales.
